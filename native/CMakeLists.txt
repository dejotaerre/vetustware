cmake_minimum_required(VERSION 3.21)
project(Vetust VERSION 0.3 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(Qt6 REQUIRED COMPONENTS Quick QuickControls2 Sql Concurrent PdfQuick)
find_package(PkgConfig REQUIRED)
pkg_check_modules(LIBZIP REQUIRED IMPORTED_TARGET libzip)
pkg_check_modules(SDL2 REQUIRED IMPORTED_TARGET sdl2)
qt_standard_project_setup(REQUIRES 6.5)
qt_add_executable(vetust main.cpp backend.cpp home.cpp backend.h displays.cpp displays.h storage.cpp storage.h emulation.cpp emulation.h archive.cpp archive.h)
qt_add_qml_module(vetust URI Vetust VERSION 1.0 QML_FILES Main.qml EmulatorDialog.qml StorageDialog.qml)
target_link_libraries(vetust PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Sql Qt6::Concurrent Qt6::PdfQuick PkgConfig::LIBZIP PkgConfig::SDL2)


enable_testing()
qt_add_executable(archive_tests archive_tests.cpp archive.cpp archive.h)
target_link_libraries(archive_tests PRIVATE Qt6::Core PkgConfig::LIBZIP)
add_test(NAME archivos_zip COMMAND archive_tests)
add_test(NAME catalogo_nativo COMMAND vetust --self-test)
set_tests_properties(catalogo_nativo PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen")

qt_add_executable(emulation_tests emulation_tests.cpp emulation.cpp emulation.h displays.cpp displays.h storage.cpp storage.h)
target_link_libraries(emulation_tests PRIVATE Qt6::Core PkgConfig::SDL2)
add_test(NAME ajustes_emuladores COMMAND emulation_tests)

find_package(Qt6 REQUIRED COMPONENTS Test)
qt_add_executable(home_tests home_tests.cpp backend.cpp backend.h home.cpp displays.cpp displays.h storage.cpp storage.h emulation.cpp emulation.h archive.cpp archive.h)
target_link_libraries(home_tests PRIVATE Qt6::Quick Qt6::QuickControls2 Qt6::Sql Qt6::Concurrent Qt6::PdfQuick Qt6::Test PkgConfig::LIBZIP PkgConfig::SDL2)
target_compile_definitions(home_tests PRIVATE VETUST_SOURCE="${CMAKE_CURRENT_SOURCE_DIR}")
add_test(NAME portada_y_navegacion COMMAND home_tests)
set_tests_properties(portada_y_navegacion PROPERTIES ENVIRONMENT "QT_QPA_PLATFORM=offscreen;QT_QUICK_BACKEND=software;QT_FORCE_STDERR_LOGGING=1")
